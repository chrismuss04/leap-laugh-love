 * @param quantity the number of shares ordered
 * @param status the order's status
 */

        UUID orderId,
        OffsetDateTime submittedAt,
        String clientEmail,
        String accountNumber,
        String symbol,
        String side,
        long quantity,
        String status
) {
}
